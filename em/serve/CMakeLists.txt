cmake_minimum_required(VERSION 3.20)
project(emv2 LANGUAGES CXX)
# EM-ENGINE-V2 (machine::engine). Layout expected next to this directory (tools/sync.sh copies it):
#   ../vendor/economic-machine/{native,products/pacifica_muse/native,markets/solana/v6}   (machine_economics + ctest)
#   ../graphloop/include                                                                 (machine::graphloop headers)
# Options:
#   EMV2_BUILDERS   ON = Arrow/Parquet builders (VM). OFF = readers, serve, tests only (skew).
#   EMV2_SANITIZE   ON = ASan + UBSan at -O1 (integration gate).
#   EMV2_EM_TESTS   ON = add economic-machine native ctest suite to this ctest run.
#   EMV2_ARROW_PREFIX  directory with include/arrow and libarrow.so[.2600] (pyarrow wheel layout), optional.
# Per-module sources are globbed by file-name convention, so modules add files without editing this file:
#   src/lib/<m>*.cpp          -> library emv2_<m>          (no Arrow; graph, ledger)
#   src/build/<m>*.cpp        -> library emv2_<m>_build    (Arrow; EMV2_BUILDERS only; tt*/bal* belong to ledger)
#   src/build/<m>_main.cpp    -> executable emv2-build-<m> (Arrow; EMV2_BUILDERS only)
#   src/build/<m>_<x>_main.cpp -> executable emv2-<m>-<x>  (Arrow; EMV2_BUILDERS only)
#   src/serve/*.cpp           -> executable emv2-serve     (links emv2_graph, emv2_ledger when present)
#   tests/<m>_*.cpp           -> test <m>_<name>           (tests/*_arrow_*.cpp only with EMV2_BUILDERS)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(EMV2_BUILDERS "Build Arrow/Parquet builders" OFF)
option(EMV2_SANITIZE "ASan + UBSan" OFF)
option(EMV2_EM_TESTS "Include economic-machine native ctest" ON)
set(EMV2_ARROW_PREFIX "" CACHE PATH "Arrow prefix in pyarrow layout (optional)")
set(EM_MACHINE_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/economic-machine/native" CACHE PATH "economic-machine native dir")
set(GRAPHLOOP_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../graphloop/include" CACHE PATH "graphloop include dir")

if(NOT EXISTS "${EM_MACHINE_NATIVE}/include/machine/economics/value.hpp")
  message(FATAL_ERROR "economic-machine native not found at ${EM_MACHINE_NATIVE} (run tools/sync.sh)")
endif()
if(NOT EXISTS "${GRAPHLOOP_INCLUDE}/machine/graphloop/durable.hpp")
  message(FATAL_ERROR "graphloop headers not found at ${GRAPHLOOP_INCLUDE} (run tools/sync.sh)")
endif()

find_package(Threads REQUIRED)
find_library(CRYPTOPP_LIB NAMES cryptopp crypto++ REQUIRED)
find_library(ZSTD_LIB NAMES zstd REQUIRED)

# Common flags: x86-64-v3 baseline (AVX2/BMI2, both skew Zen5 and VM Zen4); AVX-512 kernels use
# function-level target attributes with runtime dispatch. No -ffast-math: results must be exact
# and deterministic across hosts.
set(EMV2_FLAGS -march=x86-64-v3 -mtune=znver4 -Wall -Wextra -Wno-unused-parameter -Wno-misleading-indentation)
if(EMV2_SANITIZE)
  list(APPEND EMV2_FLAGS -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize-recover=undefined)
  set(EMV2_LINK_SAN -fsanitize=address,undefined)
endif()

add_library(emv2_core INTERFACE)
target_include_directories(emv2_core INTERFACE
  ${CMAKE_CURRENT_SOURCE_DIR}/include ${GRAPHLOOP_INCLUDE} ${EM_MACHINE_NATIVE}/include)
target_compile_options(emv2_core INTERFACE ${EMV2_FLAGS})
target_link_libraries(emv2_core INTERFACE ${CRYPTOPP_LIB} ${ZSTD_LIB} Threads::Threads ${EMV2_LINK_SAN})

# Arrow / Parquet (builders only)
if(EMV2_BUILDERS)
  add_library(emv2_arrow INTERFACE)
  if(EMV2_ARROW_PREFIX)
    target_include_directories(emv2_arrow INTERFACE ${EMV2_ARROW_PREFIX}/include)
    target_link_options(emv2_arrow INTERFACE -L${EMV2_ARROW_PREFIX} -l:libarrow.so.2600 -l:libparquet.so.2600 -Wl,-rpath,${EMV2_ARROW_PREFIX})
  else()
    find_package(Arrow REQUIRED)
    find_package(Parquet REQUIRED)
    target_link_libraries(emv2_arrow INTERFACE Arrow::arrow_shared Parquet::parquet_shared)
  endif()
  target_compile_definitions(emv2_arrow INTERFACE EMV2_WITH_ARROW=1)
endif()

enable_testing()

# ---- core tools and tests
add_executable(emv2-manifest src/tools/manifest_main.cpp)
target_link_libraries(emv2-manifest PRIVATE emv2_core)

file(GLOB CORE_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/core_*.cpp)
foreach(t ${CORE_TESTS})
  get_filename_component(n ${t} NAME_WE)
  add_executable(${n} ${t})
  target_link_libraries(${n} PRIVATE emv2_core)
  target_compile_options(${n} PRIVATE -UNDEBUG)
  add_test(NAME ${n} COMMAND ${n})
endforeach()

# ---- module libraries (no Arrow) and builders (Arrow)
set(EMV2_READER_LIBS "")
foreach(m graph ledger)
  if(m STREQUAL "ledger")
    file(GLOB LIB_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/ledger*.cpp ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/prices*.cpp)
    file(GLOB BUILD_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/build/ledger*.cpp ${CMAKE_CURRENT_SOURCE_DIR}/src/build/tt*.cpp ${CMAKE_CURRENT_SOURCE_DIR}/src/build/bal*.cpp)
  else()
    file(GLOB LIB_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/lib/${m}*.cpp)
    file(GLOB BUILD_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/build/${m}*.cpp)
  endif()
  list(FILTER BUILD_SRCS EXCLUDE REGEX ".*_main\\.cpp$")
  if(LIB_SRCS)
    add_library(emv2_${m} STATIC ${LIB_SRCS})
    target_link_libraries(emv2_${m} PUBLIC emv2_core)
    list(APPEND EMV2_READER_LIBS emv2_${m})
  endif()
  if(EMV2_BUILDERS)
    if(BUILD_SRCS)
      add_library(emv2_${m}_build STATIC ${BUILD_SRCS})
      target_link_libraries(emv2_${m}_build PUBLIC emv2_core emv2_arrow)
      if(TARGET emv2_${m})
        target_link_libraries(emv2_${m}_build PUBLIC emv2_${m})
      endif()
    endif()
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/build/${m}_main.cpp)
      add_executable(emv2-build-${m} src/build/${m}_main.cpp)
      target_link_libraries(emv2-build-${m} PRIVATE emv2_core emv2_arrow)
      if(TARGET emv2_${m}_build)
        target_link_libraries(emv2-build-${m} PRIVATE emv2_${m}_build)
      endif()
      if(TARGET emv2_${m})
        target_link_libraries(emv2-build-${m} PRIVATE emv2_${m})
      endif()
    endif()
    # extra tools: src/build/<m>_<x>_main.cpp -> emv2-<m>-<x>
    file(GLOB TOOL_MAINS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/build/${m}_*_main.cpp)
    foreach(tm ${TOOL_MAINS})
      get_filename_component(tn ${tm} NAME_WE)
      string(REGEX REPLACE "^${m}_(.*)_main$" "\\1" tx ${tn})
      add_executable(emv2-${m}-${tx} ${tm})
      target_link_libraries(emv2-${m}-${tx} PRIVATE emv2_core emv2_arrow)
      if(TARGET emv2_${m}_build)
        target_link_libraries(emv2-${m}-${tx} PRIVATE emv2_${m}_build)
      endif()
      if(TARGET emv2_${m})
        target_link_libraries(emv2-${m}-${tx} PRIVATE emv2_${m})
      endif()
    endforeach()
  endif()
  file(GLOB M_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/${m}_*.cpp)
  foreach(t ${M_TESTS})
    get_filename_component(n ${t} NAME_WE)
    if(n MATCHES "_arrow_" AND NOT EMV2_BUILDERS)
      continue()
    endif()
    add_executable(${n} ${t})
    target_link_libraries(${n} PRIVATE emv2_core)
    if(TARGET emv2_${m})
      target_link_libraries(${n} PRIVATE emv2_${m})
    endif()
    if(n MATCHES "_arrow_" AND TARGET emv2_${m}_build)
      target_link_libraries(${n} PRIVATE emv2_${m}_build)
    endif()
    target_compile_options(${n} PRIVATE -UNDEBUG)
    add_test(NAME ${n} COMMAND ${n})
  endforeach()
endforeach()

# ---- dump (verify contract): links whatever readers exist
if(EMV2_READER_LIBS)
  add_executable(emv2-dump src/tools/dump_main.cpp)
  target_link_libraries(emv2-dump PRIVATE emv2_core ${EMV2_READER_LIBS})
  foreach(l ${EMV2_READER_LIBS})
    string(TOUPPER ${l} L)
    target_compile_definitions(emv2-dump PRIVATE HAVE_${L}=1)
  endforeach()
endif()

# ---- serve
file(GLOB SERVE_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/serve/*.cpp)
if(SERVE_SRCS)
  add_executable(emv2-serve ${SERVE_SRCS})
  target_link_libraries(emv2-serve PRIVATE emv2_core ${EMV2_READER_LIBS})
  foreach(l ${EMV2_READER_LIBS})
    string(TOUPPER ${l} L)
    target_compile_definitions(emv2-serve PRIVATE HAVE_${L}=1)
  endforeach()
endif()
file(GLOB S_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/serve_*.cpp)
set(SERVE_LIB_SRCS ${SERVE_SRCS})
list(FILTER SERVE_LIB_SRCS EXCLUDE REGEX ".*/main\\.cpp$")
if(S_TESTS)
  if(SERVE_LIB_SRCS)
    add_library(emv2_serve_lib STATIC ${SERVE_LIB_SRCS})
    target_link_libraries(emv2_serve_lib PUBLIC emv2_core ${EMV2_READER_LIBS})
    foreach(l ${EMV2_READER_LIBS})
      string(TOUPPER ${l} L)
      target_compile_definitions(emv2_serve_lib PUBLIC HAVE_${L}=1)
    endforeach()
  endif()
  foreach(t ${S_TESTS})
    get_filename_component(n ${t} NAME_WE)
    add_executable(${n} ${t})
    target_link_libraries(${n} PRIVATE emv2_core ${EMV2_READER_LIBS})
    if(TARGET emv2_serve_lib)
      target_link_libraries(${n} PRIVATE emv2_serve_lib)
    else()
      foreach(l ${EMV2_READER_LIBS})
        string(TOUPPER ${l} L)
        target_compile_definitions(${n} PRIVATE HAVE_${L}=1)
      endforeach()
    endif()
    target_compile_options(${n} PRIVATE -UNDEBUG)
    add_test(NAME ${n} COMMAND ${n})
  endforeach()
endif()

# ---- economic-machine native suite (same ctest run)
if(EMV2_EM_TESTS AND EXISTS "${EM_MACHINE_NATIVE}/../products/pacifica_muse/native/learning.cpp")
  add_subdirectory(${EM_MACHINE_NATIVE} em_native)
endif()
